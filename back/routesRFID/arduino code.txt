#include <SPI.h>
#include <Ethernet.h>

// --- CONFIGURATION RÉSEAU CORRIGÉE ---
byte mac[] = { 0xDE, 0xAD, 0xBE, 0xEF, 0xFE, 0xEE }; 
IPAddress ip(172, 29, 254, 51);        
IPAddress dnsServer(172, 29, 254, 1);  
IPAddress gateway(172, 29, 254, 1);    
IPAddress subnet(255, 255, 255, 0);    

IPAddress server(172, 29, 254, 14);  
const int port = 3000;                 
EthernetClient client;                 

const int pinCapteur = 6;
int etatPrecedent = LOW; 

void setup() {
  Serial.begin(9600); 
  pinMode(pinCapteur, INPUT_PULLUP);

  // Initialisation réseau
  Ethernet.begin(mac, ip, dnsServer, gateway, subnet);
 
  delay(1500); 
  Serial.print("Système prêt ! IP de l'Arduino Capteur : ");
  Serial.println(Ethernet.localIP());

  etatPrecedent = digitalRead(pinCapteur);
}

void envoyerAlerteOuverture() {
  Serial.println("\n[ALERTE] Connexion vers 172.29.254.14:3000...");

  if (client.connect(server, port)) {
    Serial.println("[ALERTE] Connecté ! Transmission du POST...");

    String json = "{\"etat\":\"ouvert\", \"source\":\"capteur_pin6\"}";

    client.println("POST /api/capteur HTTP/1.1");
    client.println("Host: 172.29.254.14:3000");
    client.println("Content-Type: application/json");
    client.println("Connection: close");
    client.print("Content-Length: ");
    client.println(json.length());
    client.println();     
    client.println(json); 
   
    Serial.println("Payload envoyé : " + json);
    
    unsigned long strTime = millis();
    while (client.connected() && (millis() - strTime < 2000)) {
      if (client.available()) {
        char c = client.read();
        Serial.print(c);
      }
    }
    client.stop();
  } else {
    
    Serial.println("[ERREUR] Échec de connexion TCP. Vérifier pare-feu du serveur ou câblage.");
  }
}

void loop() {
  int etatActuel = digitalRead(pinCapteur);

  if (etatActuel == HIGH && etatPrecedent == LOW) {
    Serial.println("\n[EVENEMENT] Front montant détecté (Ouverture)");
    envoyerAlerteOuverture();
  }
  
  etatPrecedent = etatActuel;

  if (etatActuel == HIGH) {
    Serial.println("[CAPTEUR] Pin 6 : HIGH (Ouvert)");
  } else {
    Serial.println("[CAPTEUR] Pin 6 : LOW  (Fermé)");
  }

  delay(1000); 
}
